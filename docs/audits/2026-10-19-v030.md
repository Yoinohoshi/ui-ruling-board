## 対照と制限

- #3 は option `a-b` を `ab` に変えるだけで差分900を保持して発行できた。
- #4 は短い案の白紙をもう1段明示すると、段1=0・段2=900、拡大1件で成功した。最終段繰返しの仕様では同じ比較内容である。
- #5 は無変更の生成物では発行後も差分900を保持した。差替えは無視対象の監査用コピーだけで行った明示的な故障注入である。
- #6 は無変更の正常対照の回をコピーした器に配置して release 検査が通ることも確認した。通常の worktree の器は版とSHAが対応する。
- 合成SVGだけを再現入力として使用した。実製品の画像・内部URLはここにも Issue にも収録していない。
- 製品コード、既存テスト、設定の修正はしていない。再発防止の監査観点を AGENTS.md に反映した。
- CableRoute 用の Issue 索引生成器は本リポジトリには存在せず、本件は ui-ruling-board の Issue とこの表で追跡する。

## 再現プログラムの実行方法

以下の付録は実行した一時プログラムと実測JSONの保存であり、製品のテスト追加ではない。対象コミットの専用worktreeで `npm ci` と `npm run setup` を済ませ、下記3プログラムを各見出しのファイル名で `artifacts/audit-20261019/` に保存する。ルートから次の順で実行する。

```powershell
node artifacts/audit-20261019/probe.mjs
node artifacts/audit-20261019/controls.mjs
node artifacts/audit-20261019/provenance.mjs
```

出力先が既存なら削除・上書きせず、新しい監査用worktreeを使う。`probe.mjs` は例外も結果JSONへ記録する観測用プログラムなので、終了コードだけで成功判定しない。修正後の回帰テストにする場合は各受入条件を明示的に検証する必要がある。SHAは実行時のHEADに従う。

### probe.mjs

```javascript
import { mkdirSync, writeFileSync, readFileSync, copyFileSync, existsSync } from 'node:fs';
import { resolve, join } from 'node:path';
import { prepare } from '../../scripts/prepare-round.mjs';
import { publish } from '../../scripts/publish-round.mjs';
import { checkFolder } from '../../scripts/check-folder.mjs';
import { imageTools } from '../../scripts/image-tools.mjs';

const root = resolve('artifacts/audit-20261019');
const writeJson = (file, data) => writeFileSync(file, JSON.stringify(data, null, 2) + '\n');
const svg = (content = '') => `<svg xmlns="http://www.w3.org/2000/svg" width="400" height="240"><rect width="400" height="240" fill="white"/>${content}</svg>`;
const rect = (color, x = 40) => `<rect x="${x}" y="60" width="30" height="30" fill="${color}"/>`;

function fixture(name, options, views, shots) {
  const dir = join(root, name + '-source');
  if (existsSync(dir)) throw Error(`new output required: ${dir}`);
  mkdirSync(join(dir, 'shots'), { recursive: true });
  const description = {
    schema: 1, round: { id: name, title: 'Audit', source: 'synthetic' },
    views: views.map(id => ({ id, label: id })),
    decisions: [{ id: 'D', title: 'Decision', items: [{ id: 'D.1', title: 'Item', question: 'Which?', views,
      options: options.map((id, i) => ({ id, label: id, ...(i === 1 ? { recommended: true } : {}) })) }] }],
  };
  const captures = { 'D.1': {} };
  let serial = 0;
  for (const view of views) {
    const capture = { guard: [0, 0, 400, 240], images: {}, repeat: {} };
    captures['D.1'][view] = capture;
    for (const option of options) {
      capture.images[option] = []; capture.repeat[option] = [];
      for (const content of shots(option, view)) {
        const id = serial++;
        for (const field of ['images', 'repeat']) {
          const file = `shots/${id}-${field}.svg`;
          writeFileSync(join(dir, file), svg(content));
          capture[field][option].push(file);
        }
      }
    }
  }
  writeFileSync(join(dir, 'items.mjs'), `export const description = ${JSON.stringify(description, null, 2)};\n`);
  writeJson(join(dir, 'captures.json'), captures);
  return dir;
}

const results = [];
const run = async (name, fn) => {
  try { results.push({ name, ...(await fn()) }); }
  catch (error) { results.push({ name, error: error.message }); }
  writeJson(join(root, 'results.json'), results);
  console.log(JSON.stringify(results.at(-1)));
};

await run('filename-collision', async () => {
  const source = fixture('filename-collision', ['cur', 'a', 'a-b'], ['x', 'b-x'], o => [rect(o === 'cur' ? 'gray' : o === 'a' ? 'blue' : 'green')]);
  const output = join(root, 'filename-collision-round');
  const { def, report } = await prepare(source, output);
  const item = def.decisions[0].items[0];
  const a = item.options.find(o => o.id === 'a');
  const ab = item.options.find(o => o.id === 'a-b');
  const tools = await imageTools();
  let pixels;
  try { pixels = (await tools.compare(join(output, a.images.x[0]), join(output, ab.images.x[0]), null)).pixels; }
  finally { await tools.close(); }
  const published = join(root, 'filename-collision-published');
  await publish(output, published);
  return { originalPixels: report.comparisons.find(c => c.view === 'x' && c.pair.join(',') === 'a,a-b').pixels,
    outputPixels: pixels, collidingPaths: [ab.images.x[0], a.images['b-x'][0]], publicationAccepted: true,
    checkErrors: (await checkFolder(published, { release: true })).errors };
});

for (const repeatProposal of [false, true]) await run(repeatProposal ? 'unequal-segments-control' : 'unequal-segments', async () => {
  const name = repeatProposal ? 'unequal-segments-control' : 'unequal-segments';
  const source = fixture(name, ['cur', 'a'], ['v'], o => o === 'cur' ? ['', rect('black')] : repeatProposal ? ['', ''] : ['']);
  const { report } = await prepare(source, join(root, name + '-round'));
  return { accepted: true, comparisons: report.comparisons.map(c => ({ segment: c.segment, pixels: c.pixels })), zooms: report.zooms.length };
});

await run('stale-evidence', async () => {
  const source = fixture('stale-evidence', ['cur', 'a'], ['v'], o => [rect(o === 'cur' ? 'gray' : 'blue')]);
  const output = join(root, 'stale-evidence-round');
  const { def, report } = await prepare(source, output);
  const [current, proposed] = def.decisions[0].items[0].options;
  for (const view of def.decisions[0].items[0].views)
    for (let n = 0; n < proposed.images[view].length; n++) copyFileSync(join(output, current.images[view][n]), join(output, proposed.images[view][n]));
  const published = join(root, 'stale-evidence-published');
  await publish(output, published);
  const tools = await imageTools();
  let pixels;
  try { pixels = (await tools.compare(join(published, current.images.v[0]), join(published, proposed.images.v[0]), null)).pixels; }
  finally { await tools.close(); }
  return { evidencePixels: report.comparisons[0].pixels, publishedPixels: pixels, publicationAccepted: true };
});
```

### controls.mjs

```javascript
import { cpSync, mkdirSync, writeFileSync, readFileSync } from 'node:fs';
import { resolve, join } from 'node:path';
import { pathToFileURL } from 'node:url';
import { prepare } from '../../scripts/prepare-round.mjs';
import { publish } from '../../scripts/publish-round.mjs';
import { imageTools } from '../../scripts/image-tools.mjs';
const root = resolve('artifacts/audit-20261019');
const results = [];
for (const name of ['collision-control', 'evidence-control']) {
  const from = join(root, name === 'collision-control' ? 'filename-collision-source' : 'stale-evidence-source');
  const source = join(root, name + '-source'), output = join(root, name + '-round');
  cpSync(from, source, { recursive: true });
  if (name === 'collision-control') {
    const { description } = await import(pathToFileURL(join(source, 'items.mjs')).href);
    description.decisions[0].items[0].options[2].id = 'ab';
    const captures = JSON.parse(readFileSync(join(source, 'captures.json'), 'utf8'));
    for (const c of Object.values(captures['D.1'])) for (const field of ['images', 'repeat']) {
      c[field].ab = c[field]['a-b']; delete c[field]['a-b'];
    }
    writeFileSync(join(source, 'items.mjs'), `export const description = ${JSON.stringify(description)};\n`);
    writeFileSync(join(source, 'captures.json'), JSON.stringify(captures));
  }
  const { def, report } = await prepare(source, output);
  const options = def.decisions[0].items[0].options;
  const view = name === 'collision-control' ? 'x' : 'v';
  const indices = name === 'collision-control' ? [1, 2] : [0, 1];
  const tool = await imageTools();
  let pixels;
  try { pixels = (await tool.compare(...indices.map(i => join(output, options[i].images[view][0])), null)).pixels; }
  finally { await tool.close(); }
  await publish(output, join(root, name + '-published'));
  results.push({ name, pixels, accepted: true });
}
console.log(results);
writeFileSync(join(root, 'controls-result.json'), JSON.stringify(results, null, 2));
```

```javascript
import { mkdirSync, copyFileSync, cpSync, readFileSync, writeFileSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { spawnSync } from 'node:child_process';
import { checkFolder } from '../../scripts/check-folder.mjs';
const base = resolve('artifacts/audit-20261019');
const source = join(base, 'archive-copy-control-round');
mkdirSync(join(source, 'scripts'), { recursive: true });
cpSync('board', join(source, 'board'), { recursive: true });
copyFileSync('scripts/install.mjs', join(source, 'scripts/install.mjs'));
const boardFile = join(source, 'board/board.js');
writeFileSync(boardFile, readFileSync(boardFile, 'utf8').replace('const VERSION = "0.3.0"', 'const VERSION = "0.3.9"'));
const output = join(base, 'archive-copy-control-round-output');
const r = spawnSync(process.execPath, [join(source, 'scripts/install.mjs'), output], { encoding: 'utf8' });
for (const name of ['round.js', 'items.mjs', 'generated.json', 'evidence.json']) copyFileSync(join(base, 'evidence-control-round', name), join(output, name));
cpSync(join(base, 'evidence-control-round/images'), join(output, 'images'), { recursive: true });
const lock = JSON.parse(readFileSync(join(output, 'board.lock.json'), 'utf8'));
const parentVersion = spawnSync('git', ['show', `${lock.commit}:board/board.js`], { encoding: 'utf8' }).stdout.match(/const VERSION = "([^"]+)"/)?.[1];
const result = { installStatus: r.status, lockVersion: lock.version, recordedCommit: lock.commit, recordedCommitBoardVersion: parentVersion,
  releaseErrors: (await checkFolder(output, { release: true })).errors };
writeFileSync(join(base, 'provenance-result.json'), JSON.stringify(result, null, 2) + '\n');
console.log(result);
```
